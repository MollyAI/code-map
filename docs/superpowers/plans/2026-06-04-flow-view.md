# Flow View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the viewer's `subsystem` grouping with a **flow view** that traverses `uses`-edges forward from entry points (left→right pipeline), with hub-exclusion pruning.

**Architecture:** Phase 1 (Python) computes a `hub` flag per declaration and one deterministic candidate flow per entry point, both serialized into the JSON. Phase 2 (AI) renames/curates flows. The viewer (single HTML file) renders `flows[]` as a left→right layered DAG and supports client-side trace-from-any-node using the same pruning rules. New graph logic lives in a new, isolated `scripts/lib/flows.py` so it is unit-testable with stdlib `unittest`.

**Tech Stack:** Python 3 (stdlib only — `unittest` for tests, no third-party deps), tree-sitter (unchanged), vanilla JS + inline SVG in `viewer/index.html`.

**Testing strategy (decided):** stdlib `unittest` for all new Python logic (`tests/`, run with `python3 -m unittest discover`); the viewer is verified manually by running `analyze.py` on this repo and inspecting the browser (the project has no JS test harness and we are not adding one).

**Spec:** `docs/superpowers/specs/2026-06-04-flow-view-design.md`

```python
"""
Flow construction: turn the resolved dependency graph into per-entry-point
"flows" (forward call/dependency chains), with high-in-degree "commons" nodes
treated as non-expandable leaves so a flow never explodes into the whole graph.

Language-agnostic: operates only on Declaration objects (for hub marking) and
plain edge dicts {from, to, kind} (for traversal). Mirrors the client-side
trace in viewer/index.html — keep the pruning rules in sync.
"""
from __future__ import annotations
from collections import defaultdict, deque
from .extractors.base import Declaration


def mark_hubs(declarations: list[Declaration], percentile: float = 0.05) -> set[str]:
    """Mark the top `percentile` of declarations by in-degree as hubs.

    Hubs are the "everyone calls this" nodes (logger/util/common base class);
    flow traversal shows them but does not expand them, so a flow stays about
    what is *specific* to its path. Rank-based and gated on in_degree > 0 so a
    small repo (where the percentile boundary lands in the zero block) gets no
    hubs at all. Reads the `_in_degree` cached by core.build_graph. Sets `_hub`
    on every declaration and returns the set of hub qualified_names.
    """
    for d in declarations:
        d._hub = False  # type: ignore[attr-defined]
    nonzero = [d for d in declarations if getattr(d, "_in_degree", 0) > 0]
    # Sort by in-degree desc, then qualified_name asc for deterministic ties.
    nonzero.sort(key=lambda d: (-d._in_degree, d.qualified_name))  # type: ignore[attr-defined]
    k = int(len(nonzero) * percentile)
    hub_ids: set[str] = set()
    for d in nonzero[:k]:
        d._hub = True  # type: ignore[attr-defined]
        hub_ids.add(d.qualified_name)
    return hub_ids
```

```python
def trace_flow(seed: str, adjacency: dict, hub_ids: set[str],
               max_depth: int = 6) -> tuple[list[str], list[dict]]:
    """Forward BFS from `seed` over `adjacency` (from → [to], built from
    'uses' edges). A hub node is included but not expanded (leaf) — unless it
    is the seed itself. Each node is placed once, at its shortest depth; edges
    back to an already-placed node are omitted (keeps the pipeline a readable
    tree). Stops past `max_depth`. Returns (ordered node ids, edge dicts).
    """
    visited = {seed}
    order = [seed]
    flow_edges: list[dict] = []
    depth = {seed: 0}
    q = deque([seed])
    while q:
        u = q.popleft()
        if u != seed and u in hub_ids:
            continue                      # hub: leaf, do not expand
        if depth[u] >= max_depth:
            continue
        for v in adjacency.get(u, []):
            if v in visited:
                continue                  # already placed — omit cross/back edge
            visited.add(v)
            depth[v] = depth[u] + 1
            order.append(v)
            flow_edges.append({"from": u, "to": v})
            q.append(v)
    return order, flow_edges
```

```python
def build_flows(seeds: list[str], declarations: list[Declaration],
                edges: list[dict], hub_ids: set[str],
                max_depth: int = 6) -> list[dict]:
    """Build one deterministic candidate flow per seed (an entry-point
    qualified_name). Traverses 'uses' edges only. Skips seeds that are not
    real declarations. An entry point with no outgoing edges yields a valid
    single-node flow (an honest "no downstream calls resolved" result).
    """
    adjacency: dict = defaultdict(list)
    for e in edges:
        if e.get("kind") == "uses":
            adjacency[e["from"]].append(e["to"])

    by_qname = {d.qualified_name: d for d in declarations}
    out: list[dict] = []
    for seed in seeds:
        decl = by_qname.get(seed)
        if decl is None:
            continue
        nodes, fedges = trace_flow(seed, adjacency, hub_ids, max_depth)
        out.append({
            "id": "flow:" + seed,
            "name": decl.name,
            "description": "",
            "seed": seed,
            "nodes": nodes,
            "edges": fedges,
            "confidence": "high",
        })
    return out
```

```javascript
  state.hubIds.clear();
  for (const L of state.raw.layers) {
    for (const c of L.classes) {
      state.classById.set(c.id, c);
      if (c.hub) state.hubIds.add(c.id);
    }

// Forward BFS over 'uses' edges, mirroring scripts/lib/flows.py:trace_flow.
// Hubs are leaves (not expanded) unless they are the seed. Depth-capped.
function traceFlow(seedId) {
  const visited = new Set([seedId]);
  const nodes = [seedId];
  const edges = [];
  const depth = new Map([[seedId, 0]]);
  const queue = [seedId];
  while (queue.length) {
    const u = queue.shift();
    if (u !== seedId && state.hubIds.has(u)) continue;
    if (depth.get(u) >= state.flowMaxDepth) continue;
    const outs = (state.edgesFromIdx.get(u) || []).filter(e => e.kind === "uses");
    for (const e of outs) {
      const v = e.to;
      if (!state.classById.has(v) || visited.has(v)) continue;
      visited.add(v);
      depth.set(v, depth.get(u) + 1);
      nodes.push(v);
      edges.push({ from: u, to: v });
      queue.push(v);
    }
  }
  return { nodes, edges };
}

// Synthesize flows client-side when the JSON has none (old builds): one per
// entry-point class, traced with the same rules. New builds use raw.flows.
function synthesizeFlows() {
  const out = [];
  for (const [id, c] of state.classById) {
    const tags = Array.isArray(c.tags) ? c.tags : [];
    if (!tags.includes("entry-point")) continue;
    const { nodes, edges } = traceFlow(id);
    out.push({ id: "flow:" + id, name: c.name, description: "",
               seed: id, nodes, edges, confidence: "high" });
  }
  return out;
}

function buildFlowIndex() {

  const flows = (state.raw.flows && state.raw.flows.length)
    ? state.raw.flows : synthesizeFlows();
  for (const f of flows) state.flowsById.set(f.id, f);
  if (state.activeFlow == null || !state.flowsById.has(state.activeFlow)) {
    state.activeFlow = flows.length ? flows[0].id : null;
```

```javascript
// Left→right layered DAG layout for a single flow. Columns = hop distance from
// the seed (computed over the flow's own edges, which form a tree). Within a
// column, nodes stack vertically and the column is centred against the tallest.
function layoutFlow(flow) {
  const COL_GAP = Math.round(64 * state.fontScale);
  const ROW_GAP = LAYOUT.nodeGapY + 6;
  const PAD_X = LAYOUT.bandPadX;
  const PAD_Y = LAYOUT.bandPadTop;

  // resolve ids → data; drop any that aren't in the map
  const data = flow.nodes.map(id => state.classById.get(id)).filter(Boolean);
  if (!data.length) return { nodes: [], edges: [], width: 0, height: 0 };

  // depth via BFS over flow.edges from the seed
  const children = new Map();
  for (const e of flow.edges) {
    if (!children.has(e.from)) children.set(e.from, []);
    children.get(e.from).push(e.to);
  }
  const depth = new Map([[flow.seed, 0]]);
  const q = [flow.seed];
  while (q.length) {
    const u = q.shift();
    for (const v of (children.get(u) || [])) {
      if (!depth.has(v)) { depth.set(v, depth.get(u) + 1); q.push(v); }
    }
  }

  // bucket by depth
  const cols = new Map();   // depth -> [datum]
  for (const c of data) {
    const dp = depth.has(c.id) ? depth.get(c.id) : 0;
    if (!cols.has(dp)) cols.set(dp, []);
    cols.get(dp).push(c);
  }
  const depths = [...cols.keys()].sort((a, b) => a - b);

  // column widths + x positions
  const colW = new Map();
  for (const dp of depths) {
    colW.set(dp, Math.max(...cols.get(dp).map(c => nodeWidth(c.name, c.importance))));
  }
  const colX = new Map();
  let x = PAD_X;
  for (const dp of depths) { colX.set(dp, x); x += colW.get(dp) + COL_GAP; }
  const totalWidth = x - COL_GAP + PAD_X;

  // column heights → vertical centring
  const colH = new Map();
  for (const dp of depths) {
    const n = cols.get(dp).length;
    colH.set(dp, n * LAYOUT.nodeH + (n - 1) * ROW_GAP);
  }
  const maxColH = Math.max(...depths.map(dp => colH.get(dp)));
  const totalHeight = maxColH + PAD_Y * 2;

  const positioned = [];
  for (const dp of depths) {
    const items = cols.get(dp);
    let y = PAD_Y + (maxColH - colH.get(dp)) / 2;
    for (const c of items) {
      positioned.push({ datum: c, x: colX.get(dp), y, w: colW.get(dp), h: LAYOUT.nodeH });
      y += LAYOUT.nodeH + ROW_GAP;
    }
  }
  return { nodes: positioned, edges: flow.edges, width: totalWidth, height: totalHeight };
}

// Horizontal cubic from the right edge of `from` to the left edge of `to`.
function buildFlowEdgePath(from, to) {
  const p1 = { x: from.x + from.w, y: from.y + from.h / 2 };
  const p2 = { x: to.x,           y: to.y + to.h / 2     };
  const dx = (p2.x - p1.x) * 0.5;
  const path = document.createElementNS(NS, "path");
  path.setAttribute("class", "edge active out");
  path.setAttribute("d", `M ${p1.x} ${p1.y} C ${p1.x + dx} ${p1.y}, ${p2.x - dx} ${p2.y}, ${p2.x} ${p2.y}`);
  return path;
}
```

```javascript
    els.flowSelect.hidden = (state.grouping !== "flow");
  }
  apply(Settings.get("grouping", "layer"));   // before load(), so the first render groups correctly
  els.groupToggle.addEventListener("click", (ev) => {
    const b = ev.target.closest("button");
    if (!b) return;
    apply(b.dataset.group);
    Settings.set("grouping", state.grouping);
    state.selected = null;     // view changes; clear stale selection
    state.traceSeed = null;    // leaving/entering flow mode resets any live trace
    render();
  });
  els.flowSelect.addEventListener("change", (ev) => {
    if (ev.target.value === "__trace__") return;  // the synthetic current-trace entry
    state.traceSeed = null;
    state.activeFlow = ev.target.value;
    state.selected = null;
    render();
  });
})();
```
